cmake_minimum_required(VERSION 3.20)
project(GFDMDebugHook LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the plugin self-contained; do not require a separately installed VC runtime.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "This project currently targets Windows x86 only")
endif()
set(IMGUI vendor/imgui)
set(MINHOOK vendor/minhook)
set(COMMON src/plugin.cpp src/panel.cpp ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp
  ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/backends/imgui_impl_dx9.cpp
  ${MINHOOK}/src/buffer.c ${MINHOOK}/src/hook.c ${MINHOOK}/src/trampoline.c ${MINHOOK}/src/hde/hde32.c)
function(make_plugin target)
  add_library(${target} SHARED ${COMMON} src/exports.def)
  target_include_directories(${target} PRIVATE src ${IMGUI} ${IMGUI}/backends ${MINHOOK}/include)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_definitions(${target} PRIVATE IMGUI_USER_CONFIG="debug_imgui_config.h")
  target_link_libraries(${target} PRIVATE d3d9 bcrypt user32 gdi32 imm32 shell32)
  set_target_properties(${target} PROPERTIES PREFIX "")
  if(MINGW)
    target_link_options(${target} PRIVATE -static)
  endif()
endfunction()
make_plugin(gfdmdebug)
